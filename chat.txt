What you're describing lines up closely with how Google's actual Fast Pair protocol works, so I can build this fairly faithfully. Quick grounding on the real pieces involved:

Scanning: Fast Pair earbuds advertise BLE service UUID 0xFE2C. You filter scan results on that.
KBP (Key-based Pairing): this is Google's real term for it — it's a GATT characteristic (FE2C1234-8366-4814-8EB0-01DE32100BEA) inside the Fast Pair service. You subscribe to notifications on it, then write an encrypted 16-byte request; a responding earbud notifies back an encrypted 16-byte reply.
The catch: that handshake is AES-encrypted using a key derived from ECDH between a seeker keypair you generate and the earbud model's Anti-Spoofing Public Key — a value Google only hands out per registered model (to the OEM, or to Android via a private Google API). Without it, you can still scan, GATT-connect, and read the Model ID characteristic to confirm a device is Fast-Pair-capable — but you won't get a valid decrypted KBP response unless you have that model's key (e.g. you're the manufacturer) or the earbud already trusts you via a previously-shared Account Key.

import asyncio
from bleak import BleakScanner, BleakClient

FAST_PAIR_SERVICE = "0000fe2c-0000-1000-8000-00805f9b34fb"

MODEL_ID_CHAR = "fe2c1233-8366-4814-8eb0-01de32100bea"
KEY_BASED_PAIRING_CHAR = "fe2c1234-8366-4814-8eb0-01de32100bea"


async def scan_fast_pair_devices(timeout=8):
    print("Scanning for Fast Pair devices...\n")

    devices = await BleakScanner.discover(
        timeout=timeout,
        return_adv=True
    )

    candidates = []

    for address, (device, advertisement) in devices.items():

        service_uuids = [
            uuid.lower()
            for uuid in (advertisement.service_uuids or [])
        ]

        if FAST_PAIR_SERVICE.lower() in service_uuids:
            candidates.append(device)

            print(f"[FAST PAIR] {device.name or 'Unknown'}")
            print(f"  Address : {device.address}")
            print(f"  RSSI    : {advertisement.rssi}")
            print()

    return candidates


async def inspect_device(device):
    print(f"\nConnecting to {device.name or device.address}...")

    try:
        async with BleakClient(device) as client:

            if not client.is_connected:
                print("  Connection failed")
                return None

            print("  GATT connected")

            services = client.services

            model_char = None
            kbp_char = None

            for service in services:
                for char in service.characteristics:

                    uuid = char.uuid.lower()

                    if uuid == MODEL_ID_CHAR:
                        model_char = char

                    elif uuid == KEY_BASED_PAIRING_CHAR:
                        kbp_char = char

            model_id = None

            if model_char:
                try:
                    model_id = await client.read_gatt_char(model_char)
                    print(
                        "  Model ID:",
                        model_id.hex()
                    )
                except Exception as e:
                    print("  Could not read Model ID:", e)

            if kbp_char:
                print("  KBP characteristic found")
                print(
                    "  Properties:",
                    kbp_char.properties
                )
            else:
                print("  KBP characteristic NOT found")

            return {
                "device": device,
                "model_id": model_id,
                "kbp": kbp_char is not None,
            }

    except Exception as e:
        print("  GATT connection failed:", e)
        return None


async def main():

    devices = await scan_fast_pair_devices()

    if not devices:
        print("No Fast Pair devices found.")
        return

    print("\nFast Pair devices:")
    print("------------------")

    results = []

    for i, device in enumerate(devices):
        print(f"\n[{i}] {device.name or 'Unknown'}")
        result = await inspect_device(device)

        if result:
            results.append(result)

    print("\nCandidates:")
    print("----------")

    for i, result in enumerate(results):
        device = result["device"]

        print(
            f"[{i}] "
            f"{device.name or 'Unknown'} "
            f"({device.address}) "
            f"KBP={'yes' if result['kbp'] else 'no'}"
        )

    if not results:
        return

    try:
        choice = int(input("\nSelect earbud: "))

        if choice < 0 or choice >= len(results):
            print("Invalid selection.")
            return

        selected = results[choice]["device"]

        print(
            f"\nSelected: "
            f"{selected.name or selected.address}"
        )

        # The BLE/GATT connection is intentionally not kept here.
        # The next stage will perform the actual Fast Pair
        # authenticated KBP exchange and then establish the
        # Bluetooth audio connection.

    except ValueError:
        print("Invalid input.")


if __name__ == "__main__":
    asyncio.run(main())

/** Generates a session key for Fast Pair communication. */
function generateSharedSecret(publicKey: Buffer) {
  // generate an ephemeral elliptic curve key pair
  const ecdh = crypto.createECDH("prime256v1");
  ecdh.generateKeys();

  const generatedPublicKey = ecdh.getPublicKey();
  const generatedPrivateKey = ecdh.getPrivateKey();

  const e = new EC("p256");
  const k1 = e.keyFromPrivate(generatedPrivateKey.toString("hex"), "hex");
  const k2 = e.keyFromPublic("04" + publicKey.toString("hex"), "hex");

  const sharedSecret = Buffer.from(k1.derive(k2.getPublic()).toString("hex"), "hex");

  return {
    secret: crypto.createHash("sha256").update(sharedSecret).digest().subarray(0, 16),
    generatedPublicKey,
  };
}

/** Generates a payload for the initial pairing request. */
function generatePayload(address: string, secret: Buffer, generatedPublicKey: Buffer) {
  const nonce = crypto.randomBytes(8);

  const rawRequest = Buffer.concat([
    Buffer.from([0x00, 0x00]),
    Buffer.from(convertAddressToHex(address), "hex"),
    nonce,
  ]);

  // cipher configuration
  const cipher = crypto.createCipheriv("aes-128-ecb", secret, null);
  const decipher = crypto.createDecipheriv("aes-128-ecb", secret, null);

  const payload = Buffer.concat([encryptedRequest, generatedPublicKey.subarray(1)]);

     encrypted 16-byte request
              +
     ephemeral public key
              │
              ▼
          FE2C1234

And your generatePayload() confirms the 80-byte structure:

const rawRequest = Buffer.concat([
  Buffer.from([0x00, 0x00]),
  Buffer.from(convertAddressToHex(address), "hex"),
  nonce,
]);

That's:

2 bytes   message/header
6 bytes   Bluetooth address
8 bytes   nonce
----------------
16 bytes  plaintext

Then:

const encryptedRequest = cipher.update(rawRequest);

const payload = Buffer.concat([
  encryptedRequest,
  generatedPublicKey.subarray(1)
]);

So:

16-byte AES ciphertext
+
64-byte uncompressed P-256 public-key coordinates
=
80-byte KBP payload

That is the useful part for your Linux implementation.

One thing I would change
Your implementation uses:

crypto.createCipheriv("aes-128-ecb", secret, null)
cipher.setAutoPadding(false)

If we're implementing this ourselves, I would make the protocol version and exact message format explicit rather than blindly porting this code. That avoids accidentally implementing behavior from an old/research implementation that differs from Google's current specification.

Also, don't copy this part
This:

const ACCOUNT_KEY = Buffer.concat([
  Buffer.from([0x04]),
  Buffer.from("370beacd6f09e6f70dfe7fc5ad20a9", "hex"),
]);

is explicitly described by the source as a hardcoded account key for attacks. I would not incorporate it into your program.

Likewise, these methods are vulnerability-testing functionality rather than ordinary pairing:

testInvalidCurve()
testPairingStatePredicate()

The invalid-curve portion in particular is deliberately trying specially constructed public keys to test vulnerable implementations.

import asyncio

from bleak import BleakScanner


FAST_PAIR_SERVICE_UUID = "0000fe2c-0000-1000-8000-00805f9b34fb"


async def scan(timeout: float = 10.0):
    print(f"Scanning for Fast Pair devices for {timeout:g} seconds...\n")

    devices = await BleakScanner.discover(
        timeout=timeout,
        return_adv=True,
    )

    found = []

    for address, (device, advertisement) in devices.items():
        service_uuids = [
            uuid.lower()
            for uuid in (advertisement.service_uuids or [])
        ]

        if FAST_PAIR_SERVICE_UUID in service_uuids:
            entry = {
                "address": address,
                "name": device.name or advertisement.local_name or "Unknown",
                "rssi": advertisement.rssi,
                "manufacturer_data": advertisement.manufacturer_data,
                "service_data": advertisement.service_data,
            }

            found.append(entry)

            print(f"[+] Fast Pair device")
            print(f"    Name:    {entry['name']}")
            print(f"    Address: {entry['address']}")
            print(f"    RSSI:    {entry['rssi']}")
            print()

    if not found:
        print("No Fast Pair advertisements found.")

    return found


if __name__ == "__main__":
    asyncio.run(scan())

from bleak import BleakScanner


FAST_PAIR_UUID = "0000fe2c-0000-1000-8000-00805f9b34fb"


def normalize_uuid(uuid: str) -> str:
    return uuid.lower()


def extract_model_id(service_data: dict) -> int | None:
    """
    Extract the 24-bit Fast Pair Model ID from FE2C service data.

    Fast Pair service data begins with:
        byte 0: flags
        byte 1-3: Model ID
    """
    for uuid, data in service_data.items():
        if normalize_uuid(uuid) != FAST_PAIR_UUID:
            continue

        if len(data) < 4:
            return None

        return int.from_bytes(data[1:4], byteorder="big")

    return None


async def scan(timeout: float = 10.0):
    print(f"Scanning for Fast Pair devices for {timeout:g} seconds...\n")

    devices = await BleakScanner.discover(
        timeout=timeout,
        return_adv=True,
    )

    found = []

    for address, (device, advertisement) in devices.items():
        service_uuids = [
            normalize_uuid(uuid)
            for uuid in (advertisement.service_uuids or [])
        ]

        service_data = advertisement.service_data or {}

        is_fast_pair = (
            FAST_PAIR_UUID in service_uuids
            or any(
                normalize_uuid(uuid) == FAST_PAIR_UUID
                for uuid in service_data
            )
        )

        if not is_fast_pair:
            continue

        model_id = extract_model_id(service_data)

        entry = {
            "address": address,
            "name": device.name or advertisement.local_name or "Unknown",
            "rssi": advertisement.rssi,
            "model_id": model_id,
            "service_uuids": service_uuids,
            "service_data": service_data,
            "manufacturer_data": advertisement.manufacturer_data or {},
        }

        found.append(entry)

        print("========================================")
        print("Fast Pair device found")
        print("========================================")
        print(f"Name:       {entry['name']}")
        print(f"Address:    {entry['address']}")
        print(f"RSSI:       {entry['rssi']}")

        if model_id is not None:
            print(f"Model ID:   0x{model_id:06X} ({model_id})")
        else:
            print("Model ID:   not found")

        print("\nService UUIDs:")
        for uuid in service_uuids:
            print(f"  {uuid}")

        print("\nService data:")
        for uuid, data in service_data.items():
            print(f"  {uuid}: {bytes(data).hex()}")

class BluetoothDevice:
    def __init__(self, address: str, name: str):
        self.address = address
        self.name = name
        self.client: BleakClient | None = None

    async def connect(self) -> None:
        self.client = BleakClient(self.address)
        await self.client.connect()

    async def disconnect(self) -> None:
        if self.client and self.client.is_connected:
            await self.client.disconnect()

    @property
    def connected(self) -> bool:
        return bool(self.client and self.client.is_connected)

    async def discover_gatt(self) -> list[GattService]:
        if not self.client or not self.client.is_connected:
            raise RuntimeError("Device is not connected")

        services = []

        for service in self.client.services:
            characteristics = []

            for characteristic in service.characteristics:
                characteristics.append(
                    GattCharacteristic(
                        uuid=characteristic.uuid,
                        properties=list(characteristic.properties),
                    )
                )

            services.append(
                GattService(
                    uuid=service.uuid,
                    characteristics=characteristics,
                )
            )

        return services

import asyncio
import shutil


class BlueZError(RuntimeError):
    """Raised when a BlueZ operation fails."""


class BlueZ:
    """
    Small wrapper around bluetoothctl.

    This provides basic Bluetooth-controller operations while keeping
    bluetoothctl-specific details out of the rest of the application.
    """

    @staticmethod
    def available() -> bool:
        """Return True if bluetoothctl is installed."""
        return shutil.which("bluetoothctl") is not None

    async def _run(self, *arguments: str) -> str:
        if not self.available():
            raise BlueZError(
                "bluetoothctl was not found. "
                "Install bluez-utils."
            )

        process = await asyncio.create_subprocess_exec(
            "bluetoothctl",
            *arguments,
            stdout=asyncio.subprocess.PIPE,
            stderr=asyncio.subprocess.PIPE,
        )

        stdout, stderr = await process.communicate()

        output = stdout.decode(errors="replace").strip()
        error = stderr.decode(errors="replace").strip()

        if process.returncode != 0:
            raise BlueZError(
                error or output or
                f"bluetoothctl exited with code {process.returncode}"
            )

        return output

    async def power_on(self) -> str:
        """Power on the default Bluetooth controller."""
        return await self._run("power", "on")

    async def show(self) -> str:
        """Return information about the Bluetooth controller."""
        return await self._run("show")

    async def list_controllers(self) -> str:
        """Return the available Bluetooth controllers."""
        return await self._run("list")

from dataclasses import dataclass


@dataclass(frozen=True)
class BluetoothAudioCapabilities:
    """
    Audio capabilities exposed by a Bluetooth device.
    """

    a2dp_sink: bool = False
    hfp: bool = False
    hsp: bool = False
    le_audio: bool = False

    @property
    def has_output(self) -> bool:
        return self.a2dp_sink or self.hfp or self.hsp or self.le_audio

    @property
    def has_microphone(self) -> bool:
        return self.hfp or self.hsp or self.le_audio


def detect_from_uuids(uuids: list[str]) -> BluetoothAudioCapabilities:
    """
    Detect commonly used Bluetooth audio profiles from service UUIDs.

    This is intentionally generic. A device may expose different
    combinations of profiles.
    """

    normalized = {uuid.lower() for uuid in uuids}

    # Bluetooth SIG service UUIDs.
    A2DP_SINK = "0000110b-0000-1000-8000-00805f9b34fb"
    HFP = "0000111e-0000-1000-8000-00805f9b34fb"
    HSP = "00001108-0000-1000-8000-00805f9b34fb"

    # LE Audio uses several services rather than one universal
    # "LE Audio" UUID. We therefore don't claim LE Audio solely
    # from an arbitrary unknown UUID here.
    le_audio = False

    return BluetoothAudioCapabilities(
        a2dp_sink=A2DP_SINK in normalized,
        hfp=HFP in normalized,
        hsp=HSP in normalized,
        le_audio=le_audio,
    )

class BluetoothPairing:
    """

    Fast Pair protocol operations are intentionally kept out of this
    class.

        """
        Pair with a Bluetooth device using bluetoothctl.

        The address is supplied by the user/device discovery layer;
        this class does not contain any hard-coded device addresses.

        try:
            output = await self.bluez._run("pair", address)
        except BlueZError as exc:
            raise PairingError(str(exc)) from exc

        return output

    async def trust(self, address: str) -> str:
        """Mark a successfully paired device as trusted."""

        if not address:
            raise PairingError("Bluetooth device address is empty.")

        try:
            return await self.bluez._run("trust", address)
        except BlueZError as exc:
            raise PairingError(str(exc)) from exc

    async def remove(self, address: str) -> str:
        """Remove a device from BlueZ's paired-device database."""

        if not address:
            raise PairingError("Bluetooth device address is empty.")

        try:
            return await self.bluez._run("remove", address)

from dataclasses import dataclass

from bleak import BleakClient


@dataclass(frozen=True)
class GattCharacteristicInfo:
    uuid: str
    properties: tuple[str, ...]


@dataclass(frozen=True)
class GattServiceInfo:
    uuid: str
    characteristics: tuple[GattCharacteristicInfo, ...]


class GattClient:
    """
    Generic BLE GATT discovery.

    This class does not assume a particular earbud manufacturer,
    model, or characteristic layout.
    """

    def __init__(self, address: str):
        self.address = address
        self.client: BleakClient | None = None

    async def connect(self) -> None:
        if self.client and self.client.is_connected:
            return

        self.client = BleakClient(self.address)
        await self.client.connect()

    async def disconnect(self) -> None:
        if self.client and self.client.is_connected:
            await self.client.disconnect()

    @property
    def connected(self) -> bool:
        return bool(self.client and self.client.is_connected)

    async def discover(self) -> list[GattServiceInfo]:
        if not self.client or not self.client.is_connected:
            raise RuntimeError("BLE device is not connected")

        services: list[GattServiceInfo] = []

        for service in self.client.services:
            characteristics = tuple(
                GattCharacteristicInfo(
                    uuid=characteristic.uuid.lower(),
                    properties=tuple(characteristic.properties),
                )
                for characteristic in service.characteristics
            )

            services.append(
                GattServiceInfo(
                    uuid=service.uuid.lower(),
                    characteristics=characteristics,
                )
            )

        return services

            capabilities=capabilities,

            company_name=entry.get("company_name"),
            image_url=entry.get("image_url"),
            device_type=entry.get("device_type"),

import asyncio
from collections.abc import Awaitable, Callable

from .gatt import GattClient


ResponseCallback = Callable[[bytes], Awaitable[None] | None]


class KeyBasedPairing:
    """
    Handles transport for the Fast Pair Key-Based Pairing characteristic.

    This class deliberately does not generate or bypass Fast Pair
    authentication. It only handles sending a prepared request and
    listening for a notification response.
    """

    def __init__(self, client: GattClient, kbp_uuid: str):
        self.client = client
        self.kbp_uuid = kbp_uuid

    async def wait_for_response(
        self,
        timeout: float = 10.0,
    ) -> bytes | None:
        """
        Wait for one notification from the KBP characteristic.

        Returns:
            The notification bytes, or None if the timeout expires.
        """

        if not self.client.client or not self.client.client.is_connected:
            raise RuntimeError("BLE device is not connected")

        response_future: asyncio.Future[bytes] = (
            asyncio.get_running_loop().create_future()
        )

        def on_notification(_: int, data: bytearray) -> None:
            if not response_future.done():
                response_future.set_result(bytes(data))

        await self.client.client.start_notify(
            self.kbp_uuid,
            on_notification,
        )

        try:
            print("Waiting for KBP response...")

            try:
                response = await asyncio.wait_for(
                    response_future,
                    timeout=timeout,
                )
            except asyncio.TimeoutError:
                print(
                    f"✗ No KBP response received within "
                    f"{timeout:.0f} seconds."
                )
                return None

            print("✓ KBP response received!")
            print(f"  Length: {len(response)} bytes")

            return response

        finally:
            try:
                await self.client.client.stop_notify(
                    self.kbp_uuid
                )
            except Exception:
                pass

    async def send_request(
        self,
        payload: bytes,
        timeout: float = 10.0,
    ) -> bytes | None:
        """
        Send a prepared KBP payload and wait for the first response.
        """

        if not self.client.client or not self.client.client.is_connected:
            raise RuntimeError("BLE device is not connected")

        print("Sending Key-Based Pairing request...")

        response_task = asyncio.create_task(
            self.wait_for_response(timeout)
        )

        try:
            await self.client.client.write_gatt_char(
                self.kbp_uuid,
                payload,
                response=False,
            )
        except Exception:
            response_task.cancel()

            try:
                await response_task
            except asyncio.CancelledError:
                pass

            raise

        return await response_task

from collections.abc import Awaitable, Callable

from bleak import BleakClient


KBPResponseCallback = Callable[[bytes], Awaitable[None] | None]


class KBPClient:
    """
    Generic Fast Pair Key-Based Pairing transport.

    This class only handles the BLE/GATT transport:
    - subscribe to KBP notifications
    - write a KBP packet
    - report responses

    It does not contain manufacturer-specific logic.
    """

    def __init__(
        self,
        client: BleakClient,
        characteristic_uuid: str,
    ) -> None:
        self.client = client
        self.characteristic_uuid = characteristic_uuid.lower()

    async def listen(
        self,
        callback: KBPResponseCallback,
    ) -> None:
        if not self.client.is_connected:
            raise RuntimeError("BLE device is not connected")

        async def notification_handler(
            _sender: object,
            data: bytearray,
        ) -> None:
            payload = bytes(data)

            print(
                f"KBП response received: "
                f"{len(payload)} bytes"
            )

            result = callback(payload)

            if result is not None:
                await result

        await self.client.start_notify(
            self.characteristic_uuid,
            notification_handler,
        )

    async def stop_listening(self) -> None:
        if not self.client.is_connected:
            return

        try:
            await self.client.stop_notify(
                self.characteristic_uuid
            )
        except Exception:
            pass

    async def write(self, payload: bytes) -> None:
        if not self.client.is_connected:
            raise RuntimeError("BLE device is not connected")

        await self.client.write_gatt_char(
            self.characteristic_uuid,
            payload,
            response=True,
        )

from collections.abc import Awaitable, Callable

from bleak import BleakClient


KBPResponseCallback = Callable[[bytes], Awaitable[None] | None]


class KBPClient:
    """
    Generic Fast Pair Key-Based Pairing GATT transport.

    This class does not contain manufacturer-specific logic.
    It only handles the GATT characteristic used for KBP transport.
    """

    def __init__(
        self,
        client: BleakClient,
        characteristic_uuid: str,
        properties: tuple[str, ...],
    ) -> None:
        self.client = client
        self.characteristic_uuid = characteristic_uuid.lower()
        self.properties = tuple(
            property_name.lower()
            for property_name in properties
        )

    @property
    def can_write(self) -> bool:
        return (
            "write" in self.properties
            or "write-without-response" in self.properties
        )

    @property
    def can_notify(self) -> bool:
        return (
            "notify" in self.properties
            or "indicate" in self.properties
        )

    async def listen(
        self,
        callback: KBPResponseCallback,
    ) -> None:
        if not self.client.is_connected:
            raise RuntimeError(
                "BLE device is not connected"
            )

        if not self.can_notify:
            raise RuntimeError(
                "KBP characteristic does not support "
                "notifications or indications"
            )

        async def notification_handler(
            _sender: object,
            data: bytearray,
        ) -> None:
            payload = bytes(data)

            print(
                f"KBP response received: "
                f"{len(payload)} bytes"
            )

            result = callback(payload)

            if result is not None:
                await result

        await self.client.start_notify(
            self.characteristic_uuid,
            notification_handler,
        )

    async def stop_listening(self) -> None:
        if not self.client.is_connected:
            return

        try:
            await self.client.stop_notify(
                self.characteristic_uuid
            )
        except Exception:
            pass

    async def write(self, payload: bytes) -> None:
        if not self.client.is_connected:
            raise RuntimeError(
                "BLE device is not connected"
            )

        if not self.can_write:
            raise RuntimeError(
                "KBP characteristic does not support writing"
            )

        await self.client.write_gatt_char(
            self.characteristic_uuid,
            payload,
            response="write" in self.properties,
        )
